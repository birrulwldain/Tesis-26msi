% Slide 9
\begin{frame}{Output Model \& Parameter Plasma}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\large 1. Prediksi Konsentrasi $\hat{\mathbf{c}}$}
        \fontsize{7.5pt}{9pt}\selectfont
        Vektor konsentrasi diprediksi pada simpleks tertutup ($\sum_z \hat{c}_z = 1$). \\
        \vspace{0.4em}
        \textbf{Loss Function:}
        \begin{equation*}
        \mathcal{L}_{\mathrm{conc}} = \mathcal{L}_{\mathrm{KL}}(\mathbf{c} \parallel \hat{\mathbf{c}}) + \mu \mathcal{L}_{\mathrm{CLR}}(\mathbf{c}, \hat{\mathbf{c}})
        \end{equation*}
      \end{block}
      \begin{block}{\large 2. Parameter Plasma ($T_e, n_e$)}
        \fontsize{7.5pt}{9pt}\selectfont
        Suhu eksitasi dan kerapatan Stark diestimasi secara simultan. \\
        \vspace{0.4em}
        \textbf{Loss Function:}
        \begin{equation*}
        \mathcal{L}_{\mathrm{phys}} = (\hat{T}_e - T_e)^2 + w_{n_e} (\hat{n}_e - n_e)^2
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\large 3. Rekonstruksi Spektrum Murni $\hat{\mathbf{I}}_{\mathrm{pred}}$}
        \fontsize{7.5pt}{9pt}\selectfont
        Decoder merekonstruksi garis emisi individu yang terdekomposisi. \\
        \vspace{0.4em}
        \textbf{Loss Function:}
        \begin{equation*}
        \mathcal{L}_{\mathrm{spec}} = \mathcal{L}_{\mathrm{MSE\text{-}terbobot}} + \beta_{\mathrm{grad}}\mathcal{L}_{\mathrm{grad}} + w_{\mathrm{sam}}\mathcal{L}_{\mathrm{SAM}} + \lambda_{L_1}\mathcal{L}_{L_1}
        \end{equation*}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Prosedur: Pelatihan Adversarial \& Loss Physics (Tahap 3)}
  \begin{columns}[t]
    \begin{column}{0.58\textwidth}
      \begin{block}{\large Pengekang Superposisi Spektral $\mathcal{L}_{\mathrm{cons}}$}
        \fontsize{7.5pt}{9pt}\selectfont
        Mengawal kesesuaian linear antara jumlah spektrum unsur murni prediksi dengan spektrum campuran aktual:
        \begin{equation*}
        \mathcal{L}_{\mathrm{cons}} = \frac{1}{B \cdot L} \sum_{i=1}^{B} \sum_{j=1}^{L} \left( \left[ \sum_{e=1}^{E} \hat{I}_{\mathrm{pred}, ie}(\lambda_j) \right] - x_i(\lambda_j) \right)^2
        \end{equation*}
        Secara fisis menghindari prediksi puncak emisi fiktif atau klaim ganda (*double-counting*).
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{block}{\large Fungsi Kerugian Total}
        \fontsize{7.5pt}{9pt}\selectfont
        Model dioptimalkan secara multi-task pada data pre-training:
        \begin{align*}
        \mathcal{L}_{\mathrm{total}} = & w_{\mathrm{conc}}\mathcal{L}_{\mathrm{KL}} + w_{\mathrm{clr}}\mathcal{L}_{\mathrm{CLR}} \\
        & + w_{\mathrm{spec}}\mathcal{L}_{\mathrm{spec}} + w_{\mathrm{phys}}\mathcal{L}_{\mathrm{phys}} \\
        & + \lambda_{\mathrm{cons}}\mathcal{L}_{\mathrm{cons}}
        \end{align*}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
