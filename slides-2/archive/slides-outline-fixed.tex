\begin{frame}{Motivation: Why LIBS is hard}
  \begin{itemize}
    \item Overlapping lines; matrix effects
    \item High-resolution spectra → compute bottleneck
    \item Need fast, calibration-free screening
  \end{itemize}
  % Visual placeholder (replace or remove)
  \vspace{4pt}\fbox{\parbox{0.96\textwidth}{Visual: LIBS spectrum with overlapping peaks (placeholder)}}
\end{frame}

% 1:30–2:15
\begin{frame}{Method: Informer for LIBS}
  \begin{itemize}
    \item ProbSparse Self-Attention: \(\mathcal O(L\log L)\)
    \item 2-layer Informer → efficient, avoids overfitting
    \item MultiLabelFocalLoss for class imbalance
  \end{itemize}
  \vspace{4pt}\fbox{\parbox{0.96\textwidth}{Visual: Tiny encoder diagram + ProbSparse callout (placeholder)}}
\end{frame}

% 2:15–3:00
\begin{frame}{Data: Physics-based synthetic spectra}
  \begin{itemize}
    \item Saha–Boltzmann plasma modeling; varied conditions
    \item Synthetic-only training for coverage \& control
    \item Evaluation on target dataset
  \end{itemize}
  \vspace{4pt}\fbox{\parbox{0.96\textwidth}{Visual: Pipeline (physics → synthetic → training) placeholder}}
\end{frame}

% 3:00–4:00
\begin{frame}{Results: Accuracy vs. depth}
  \begin{itemize}
    \item 2-layer best vs. \{1,4,8\} layers (F1 = 0.94, 0.92, 0.93, 0.91)
    \item Sample efficiency: 1k samples competitive with 10k
    \item Speed: \(0.3\,\text{ms}/\text{spectrum}\) vs. \(15\,\text{ms}\) (CNN)
  \end{itemize}
  \vspace{4pt}\fbox{\parbox{0.96\textwidth}{Visual: Bar chart of F1 scores + timing (placeholder)}}
\end{frame}

% 4:00–5:30
\begin{frame}{Herbal screening case study}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Success}
      \begin{itemize}
        \item Major: Fe, Ca, Mg (\textcolor{green}{✓})
        \item Trace: Cu, Zn (\textcolor{green}{✓})
        \item Fast 2-class triage
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Limits}
      \begin{itemize}
        \item Precision hit by peak overlaps
        \item Needs domain priors / grouping
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% 6:15–7:00
\begin{frame}{Takeaways}
  \begin{itemize}
    \item Informer (\(\mathcal O(L\log L)\)) enables scalable LIBS
    \item Works on complex herbal matrices (major/minor/trace)
    \item Ready for screening; next: precision via priors
  \end{itemize}
\end{frame}
